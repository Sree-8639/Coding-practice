class Person:
    def __init__(self, name, age):
        self.name = name
        self.age = age
​
a = Person("Max", 30)
b = Person("Emilia", 25)
​
print(a.name, a.age) 
print(b.name, b.age)
class Dog:
    def __init__(self, name, breed="Mixed", age=1):
        self.name = name

        self.age = age
